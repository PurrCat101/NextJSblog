```jsx
  e.preventDefault();
  setError("");
  setIsLoading(true);

  }
};
```

### การกรองด้วยวันที่ (Date Filter)

เมื่อเลือกวันที่ setStartDate หรือ setEndDate จะทำการเรียก function fetchRepairTasks โดยจะทำการ filter ข้อมูลโดยใช้วันที่ที่เลือก

<CodeHead lang="jsx" />
```jsx
<DatePicker
  selected={startDate}
  onChange={setStartDate}
  placeholderText="Start date"
  className="border rounded-lg p-2"
/>
```

โดย query ที่ใช้ในการ filter ข้อมูลด้วยวันที่จะมีการเปลี่ยนแปลงตามวันที่ที่เลือก

1. กรองข้อมูล startDate เป็นต้นไป (gte) greater than or equal
2. กรองข้อมูล endDate ถึงวันถัดไป (lt) less than

เมื่อเลือกวันที่ endDate จะทำการบวก 1 วันเพื่อเพื่อให้ระบบทำการกรองข้อมูลถึงวันสุดท้ายที่เลือกได้เนื่องจาก query ด้วย less than คือหมายถึงน้อยกว่าวันที่เลือก(ปรับเป็น less than or equal ก็ได้จะได้ไม่ต้องบวก 1 วัน)

<CodeHead lang="jsx" />
```jsx
if (startDate) {
  query = query.gte(dateField, startDate.toISOString());
}
if (endDate) {
  const nextDay = new Date(endDate);
  nextDay.setDate(nextDay.getDate() + 1);
  query = query.lt(dateField, nextDay.toISOString());
}
```

```jsx
const handleSparePartQuantityChange = (sparePartId, quantity) => {
  setSelectedSpareParts(
    selectedSpareParts.map((sp) =>
      sp.spare_part_id === sparePartId ? { ...sp, quantity } : sp
    )
  );
};
```

3. **การลบอะไหล่** (handleRemoveSparePart):

<CodeHead lang="jsx" />

```jsx
const handleRemoveSparePart = (sparePartId) => {
```

## ระบบแจ้งเตือนแบบ Real-time ด้วย React และ Supabase

### ออกแบบโครงสร้างฐานข้อมูล

สร้างตาราง `notifications` ใน Supabase ที่จะเก็บข้อมูลการแจ้งเตือน:

<CodeHead lang="sql" />

```sql
CREATE TABLE notifications (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    recipient_id UUID REFERENCES auth.users(id) NOT NULL,
    title TEXT NOT NULL,
    message TEXT NOT NULL,
    type VARCHAR(50) NOT NULL,
    read BOOLEAN DEFAULT false,
    task_id UUID REFERENCES repair_tasks(id),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    for_role VARCHAR(50)
);

-- สร้าง index เพื่อเพิ่มประสิทธิภาพการค้นหา
CREATE INDEX idx_notifications_recipient ON notifications(recipient_id);
CREATE INDEX idx_notifications_task ON notifications(task_id);
```

#### คำอธิบาย Field ต่างๆ:

- `id`: รหัสเฉพาะของการแจ้งเตือน
- `recipient_id`: รหัสผู้รับการแจ้งเตือน
- `title`: หัวข้อการแจ้งเตือน
- `message`: รายละเอียดการแจ้งเตือน
- `type`: ประเภทการแจ้งเตือน (new_task, status_change, task_assigned)
- `read`: สถานะการอ่าน (true/false)
- `task_id`: รหัสอ้างอิงถึงงานซ่อม
- `created_at`: เวลาที่สร้าง
- `updated_at`: เวลาที่อัพเดทล่าสุด
- `for_role`: บทบาทที่สามารถเห็นการแจ้งเตือน (เช่น admin)

### การทำงานของระบบ

#### ประเภทการแจ้งเตือน (Event Types)

ระบบรองรับการแจ้งเตือน 3 ประเภทหลัก:

**แจ้งเตือนงานใหม่** (NEW_TASK)
เมื่อมีการสร้างงานซ่อมใหม่
แจ้งเตือนไปยังผู้ดูแลระบบ

**แจ้งเตือนการเปลี่ยนสถานะ** (STATUS_CHANGE)
เมื่อสถานะงานเปลี่ยนเป็น "เสร็จสิ้น" หรือ "ไม่สามารถซ่อมได้"
แจ้งเตือนไปยังผู้ดูแลระบบ

**แจ้งเตือนการมอบหมายงาน** (TASK_ASSIGNED)
เมื่อมีการมอบหมายงานให้ช่างซ่อม
แจ้งเตือนไปยังช่างที่ได้รับมอบหมาย

### การจัดการแจ้งเตือน (Notification Service)

file `NotificationService.js`

<CodeHead lang="js" />

```javascript
class NotificationService {
  // ประเภทการแจ้งเตือน
  static Types = {
    NEW_TASK: "new_task",
    STATUS_CHANGE: "status_change",
    TASK_ASSIGNED: "task_assigned",
  };

  // สร้างการแจ้งเตือนใหม่
  async createNewTaskNotification(creatorId, deviceName, issue, taskId) {
    return this.createNotification({
      recipient_id: creatorId,
      title: "งานซ่อมใหม่",
      message: `มีการเพิ่มงานซ่อมใหม่: ${deviceName} - ${issue}`,
      type: NotificationTypes.NEW_TASK,
      task_id: taskId,
      for_role: "admin",
    });
  }

  // แจ้งเตือนการเปลี่ยนสถานะ
  async createStatusChangeNotification(
    changerId,
    deviceName,
    issue,
    newStatus,
    changerName,
    taskId
  ) {
    const status = newStatus === "completed" ? "เสร็จสิ้น" : "ไม่สามารถซ่อมได้";
    return this.createNotification({
      recipient_id: changerId,
      title: "สถานะงานเปลี่ยนแปลง",
      message: `งานซ่อม ${deviceName} - ${issue} ถูกเปลี่ยนสถานะเป็น${status} โดย ${changerName}`,
      type: NotificationTypes.STATUS_CHANGE,
      task_id: taskId,
      for_role: "admin",
    });
  }

  // แจ้งเตือนการมอบหมายงาน
  async createTaskAssignedNotification(
    technicianId,
    deviceName,
    issue,
    taskId,
    assignerName
  ) {
    return this.createNotification({
      recipient_id: technicianId,
      title: "ได้รับมอบหมายงานใหม่",
      message: `คุณได้รับมอบหมายให้ซ่อม ${deviceName} - ${issue} โดย ${assignerName}`,
      type: NotificationTypes.TASK_ASSIGNED,
      task_id: taskId,
    });
  }
}
```

```jsx
useEffect(() => {
  if (user) {
    // Subscribe to realtime notifications
    const channel = supabase
      .channel("notifications")
      .on(
        "postgres_changes",
        {
          event: "INSERT",
          schema: "public",
          table: "notifications",
        },
        (payload) => {
          // ตรวจสอบสิทธิ์การเข้าถึง

            setUnreadCount((prev) => prev + 1);
          }
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }
}, [user]);
```

## การจัดการ Session ใน React ด้วย Supabase

### หลักการทำงานของ Session Management

Session management เป็นส่วนสำคัญในการพัฒนาแอปพลิเคชัน web ที่ต้องมีระบบ authentication โดยในโปรเจคนี้เราใช้ Supabase
ซึ่งจะช่วยจัดการเรื่อง session ให้เราได้อย่างมีประสิทธิภาพ

### การทำงานของ Session ใน Supabase

Supabase จะจัดการ session ผ่าน JWT (JSON Web Token) โดยมีขั้นตอนดังนี้:

1. เมื่อผู้ใช้ทำการ login สำเร็จ Supabase จะสร้าง JWT token
2. Token นี้จะถูกเก็บไว้ใน localStorage ของ browser โดยอัตโนมัติ
3. Token จะถูกใช้ในการยืนยันตัวตนสำหรับทุกๆ request ที่ส่งไปยัง Supabase
4. Token มีอายุการใช้งาน เมื่อหมดอายุผู้ใช้จะต้อง login ใหม่

### การจัดการ User Session ในโค้ด

ในโปรเจคของเรา เราจัดการ session ผ่านสองส่วนหลักๆ:

### การตรวจสอบ Session

<CodeHead lang="jsx" />

```jsx
// ใน App.jsx
useEffect(() => {
  const checkUser = async () => {
    try {
      // ตรวจสอบ user จาก localStorage ก่อน
      const storedUser = localStorage.getItem("user");
      if (storedUser) {
        setUser(JSON.parse(storedUser));
      }

      // ตรวจสอบ session จาก Supabase
      const currentUser = await getCurrentUser();
      if (currentUser) {
        const userData = {
          ...currentUser,
          role: currentUser.user_metadata?.role || "user",
        };
        setUser(userData);
        localStorage.setItem("user", JSON.stringify(userData));
      } else if (!storedUser) {
        setUser(null);
        localStorage.removeItem("user");
      }
    } catch (error) {
      console.error("Error checking user session:", error);
    }
  };
}, []);
```

```jsx
// ฟังก์ชัน Login
const handleLogin = (userData) => {
  setUser(userData);
  localStorage.setItem("user", JSON.stringify(userData));
};

// ฟังก์ชัน Logout
const handleLogout = () => {
  setUser(null);
  localStorage.removeItem("user");
};
```

### การใช้งาน Supabase Client

เราใช้ Supabase Client ในการจัดการ authentication:

<CodeHead lang="jsx" />

```jsx
// lib/supabase.js
import { createClient } from "@supabase/supabase-js";

const supabaseUrl = import.meta.env.VITE_SUPABASE_URL;
const supabaseKey = import.meta.env.VITE_SUPABASE_ANON_KEY;

export const supabase = createClient(supabaseUrl, supabaseKey);
```

#### ข้อดีของการใช้ Supabase สำหรับ Session Management

**ความปลอดภัย**: Supabase จัดการเรื่อง token และการเข้ารหัสให้โดยอัตโนมัติ
**ง่ายต่อการใช้งาน**: API ที่ใช้งานง่าย ไม่ต้องเขียนโค้ดซับซ้อน
**Real-time Support**: สามารถใช้ real-time features ได้โดยใช้ token เดียวกัน
**Auto-refresh**: Token จะถูก refresh อัตโนมัติเมื่อใกล้หมดอายุ

### การจัดการ State ใน React

ในแอพของเรา เราใช้ React state เพื่อเก็บข้อมูล user:

<CodeHead lang="jsx" />

```jsx
const [user, setUser] = useState(null);
const [loading, setLoading] = useState(true);
```

- `user` state เก็บข้อมูลของผู้ใช้ที่ login อยู่
- `loading` state ใช้แสดง loading state ระหว่างตรวจสอบ session

### การ Redirect ตาม Role

- admin redirect to home page
- non-admin redirect to repair-tasks page

<CodeHead lang="jsx" />

```jsx
const getDefaultPath = () => {
  return user?.role === "admin" ? "/" : "/repair-tasks";
};
```

### Best Practices

**เช็ค localStorage และ Supabase Session**: ตรวจสอบทั้งสองที่เพื่อความรวดเร็วและความถูกต้อง
**Error Handling**: จัดการ error ที่อาจเกิดขึ้นในทุกขั้นตอน
**Loading State**: แสดง loading state ระหว่างตรวจสอบ session
**Role-based Access**: จัดการการเข้าถึงตาม role ของผู้ใช้
**Clean Up**: ลบข้อมูล user ออกจาก localStorage เมื่อ logout
